Quick Start Example
# Run a cardiac perturbation via the Zenith API
import requests

response = requests.post(
    "https://niluslab.com/api/v1/predict/perturbation",
    headers={"X-API-Key": "zk_live_your_key_here"},
    json={
        "baseline_cell_type": "ventricular_myocyte",
        "perturbation_factors": {"GATA4": 1.5, "SIRT1": 2.0},
        "census_filter": "tissue_general == 'heart' and disease == 'normal'"
    }
)

# Returns job_id ÔåÆ poll status ÔåÆ download .h5ad
job_id = response.json()["job_id"]
print(f"Job launched: {job_id}")
