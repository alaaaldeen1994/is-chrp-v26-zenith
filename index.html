function plotResultECG(ecgData) {
  const canvas = document.getElementById('ecg-canvas-result');
  if (!canvas) return;
  const ctx = canvas.getContext('2d');
  
  // 1. Draw ECG Grid Paper Background (Light Pink/Cream)
  ctx.fillStyle = '#fff5f5';
  ctx.fillRect(0, 0, canvas.width, canvas.height);
  
  // 2. Draw Small Grid Lines (1mm equivalent - every 4px)
  ctx.strokeStyle = 'rgba(239, 68, 68, 0.12)'; // Light red/pink grid
  ctx.lineWidth = 0.5;
  for (let x = 0; x < canvas.width; x += 4) {
    if (x % 20 !== 0) {
      ctx.beginPath();
      ctx.moveTo(x, 0);
      ctx.lineTo(x, canvas.height);
      ctx.stroke();
    }
  }
  for (let y = 0; y < canvas.height; y += 4) {
    if (y % 20 !== 0) {
      ctx.beginPath();
      ctx.moveTo(0, y);
      ctx.lineTo(canvas.width, y);
      ctx.stroke();
    }
  }
  
  // 3. Draw Large Grid Lines (5mm equivalent - every 20px)
  ctx.strokeStyle = 'rgba(239, 68, 68, 0.32)'; // Stronger red/pink grid
  ctx.lineWidth = 1.0;
  for (let x = 0; x < canvas.width; x += 20) {
    ctx.beginPath();
    ctx.moveTo(x, 0);
    ctx.lineTo(x, canvas.height);
    ctx.stroke();
  }
  for (let y = 0; y < canvas.height; y += 20) {
    ctx.beginPath();
    ctx.moveTo(0, y);
    ctx.lineTo(canvas.width, y);
    ctx.stroke();
  }
  
  // 4. Plot the ECG Rhythm Trace (Deep Navy Ink)
  ctx.strokeStyle = '#0f172a';
  ctx.lineWidth = 2.0;
  ctx.lineJoin = 'round';
  ctx.lineCap = 'round';
  ctx.beginPath();
  
  const stepX = canvas.width / ecgData.length;
  ecgData.forEach((val, i) => {
    const x = i * stepX;
    const y = canvas.height - (val * canvas.height * 0.8) - 10;
    if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
  });
  ctx.stroke();
}
